! Uses conventional network, popup, popunder and cosmetic filters.

! =====================================================
! XVIDEOS — page ads, prerolls and advertising tabs
! =====================================================

! Known first-party advertising and tracking paths
||xvideos.com/zoneload/$domain=xvideos.com|xvideos2.com
||xvideos.com/picserror/$domain=xvideos.com|xvideos2.com
||xvideos.com/lvcsm/abck-banners/

! Known advertising providers
||orbsrv.com^$third-party,domain=xvideos.com|xvideos2.com
||a.orbsrv.com/ad-provider.js$script,third-party,domain=xvideos.com|xvideos2.com
||pemsrv.com^$third-party,domain=xvideos.com|xvideos2.com
||pemsrv.com^$popup,domain=xvideos.com|xvideos2.com
||pemsrv.com^$popunder,domain=xvideos.com|xvideos2.com

! In-player VAST advertising selection
/vast.php?idzone=$domain=xvideos.com|xvideos2.com

! Known advertising-media paths only.
! Do not replace these with a generic *.mp4 rule.
! Generic third-party MP4 blocking breaks legitimate
! XVideos thumbnail previews and can interfere with playback.
||aucdn.net/library/*.mp4$media,third-party,domain=xvideos.com|xvideos2.com
||bkcdn.net/library/*.mp4$media,third-party,domain=xvideos.com|xvideos2.com

! Block click-triggered advertising tabs and popunders
*$popup,domain=xvideos.com|xvideos2.com
*$popunder,domain=xvideos.com|xvideos2.com

! Hide known advertising links and containers.
! Do NOT hide #video-right: it can contain legitimate
! player controls such as quality/subtitle/audio settings.
xvideos.com,xvideos2.com##a[href*="/go.php?"][href*="&ad="]
xvideos.com,xvideos2.com###video-sponsor-links
xvideos.com,xvideos2.com###ad-footer
xvideos.com,xvideos2.com##.player-video.xv-cams-block
xvideos.com,xvideos2.com##.thumb-ad
xvideos.com,xvideos2.com##.video-ad
xvideos.com,xvideos2.com##.remove-ads > *
xvideos.com,xvideos2.com##.e-banner-game
xvideos.com,xvideos2.com##.top-top
xvideos.com,xvideos2.com##[href*="base64"]
xvideos.com,xvideos2.com##[href*="data:"]

! =====================================================
! HENTAIHAVEN — ads, VAST requests and advertising tabs
! =====================================================

! Known external advertising and redirect services
||adtng.com^$third-party,domain=hentaihaven.xxx
||havenclick.com^$third-party,domain=hentaihaven.xxx
||ads-twitter.com^$third-party,domain=hentaihaven.xxx
||cdn.pornhaven.ai/design/ads/$domain=hentaihaven.xxx

! Block JavaScript belonging to the first-party
! advertising/interstitial Script Manager plugin.
!
! Use the plugin path rather than one exact JS filename so
! renamed/versioned Script Manager files cannot attach a
! popup handler to every tap.
!
! Do NOT block the separate player-logic plugin.
||hentaihaven.xxx/wp-content/plugins/script-manager/*.js$script

! Older script-manager infrastructure
||hentaihaven.icu/wp-content/plugins/sscript-manager/

! Block VAST and advertising-delivery requests
||havenclick.com/www/delivery/$xmlhttprequest,domain=hentaihaven.xxx

! Block known advertising popup/popunder destinations.
! These are intentionally destination-specific rather than
! *$popup/*$popunder for every HentaiHaven navigation.
||havenclick.com^$popup,domain=hentaihaven.xxx
||havenclick.com^$popunder,domain=hentaihaven.xxx
||adtng.com^$popup,domain=hentaihaven.xxx
||adtng.com^$popunder,domain=hentaihaven.xxx

! Hide known advertising frames and containers
hentaihaven.xxx##iframe[src*="adtng.com"]
hentaihaven.xxx##.script_manager_video_master
hentaihaven.xxx##.vrav_a_pc
hentaihaven.xxx##.popecrazy
hentaihaven.xxx##.hhzt
hentaihaven.xxx##.code-block
hentaihaven.xxx##.aloading > div[class*=" col-md-zarat "]:has(> .related-reading-wrap > .widget-thumbnail > ins[data-revive-zoneid])
